        // pin that button is connected to
        const int buttonPin = 2;
        // pin for red LED (PWM, used for fading)
        const int ledRed = 11;
        // pin for green LED (digital)
        const int ledGreen = 12;
        // pin for blue LED (digital)
        const int ledBlue = 13;

        // how bright the fading LED is (0-255)
        int brightness = 0;
        // how much to fade the brightness by, each step
        int fadeAmount = 5;

        // setup runs once when the Arduino powers on or resets
        void setup() {
          // set the button pin as an input to read it!
          pinMode(buttonPin, INPUT);
          // declare pin 11 (red LED) to be an output
          pinMode(ledRed, OUTPUT);
          // declare pin 12 (green LED) to be an output
          pinMode(ledGreen, OUTPUT);
          // declare pin 13 (blue LED) to be an output
          pinMode(ledBlue, OUTPUT);
        }

        // loop runs over and over forever
        void loop() {
          // read button: HIGH when pressed, LOW when released
          int buttonState = digitalRead(buttonPin);

          // list of LED pins for for loop to run through
          int ledPins[3] = {ledRed, ledGreen, ledBlue};

          // check if button is being pressed
          if (buttonState == HIGH) {
            // for loop
            for (int i = 0; i < 3; i++) {
              // turn current LED on
              digitalWrite(ledPins[i], HIGH);
              // delay to keep light on visibly
              delay(100);
              // turn current LED back off
              digitalWrite(ledPins[i], LOW);
            }

            // for loops runs backwards
            for (int i = 2; i >= 0; i--) {
              // turn current LED on
              digitalWrite(ledPins[i], HIGH);
              // delay to keep light on visibly
              delay(100);
              // turn current LED back off
              digitalWrite(ledPins[i], LOW);
            }
          }

          // do the fade
          else {
            // make sure green LED is off
            digitalWrite(ledGreen, LOW);
            // make sure blue LED is off
            digitalWrite(ledBlue, LOW);

            // set red LED to current brightness
            analogWrite(ledRed, brightness);
            // update brightness for next time
            brightness += fadeAmount;
            // reverse fade direction at ends of fade:
            if (brightness <= 0 || brightness >= 255) {
              // reverse fade direction
              fadeAmount = -fadeAmount;
            }
            
            // make fade visible + smooth
            delay(30);
          } 
        }

2: If all your LEDs were on simultaneously, what would the total current draw be? Is this above or below the total Arduino current draw limits? If all your LEDs were on, how long would your circuit run if powered by a 1200 mAh battery?

If all my LEDs were on simultaneously, and if I used the recommended 20 mA per LED, it would draw about 60mA of total current, since I have 3 LEDs. My actual circuit is different because each LED has its own resistor.

Using a 5 V supply, Red & Green (2.0 V, 100 Ω): (5 - 2.0) / 100 = 30 mA. Blue (3.3 V, 100 Ω): (5 - 3.3) / 100 ≈ 17 mA

The total current is about 77 mA. This is below the total Arduino current draw limits, which is 150 mA total. If all my LEDs were on, my circuit would likely run for 15.6 hours. Since 1200 mAh is 1200 mA per hour, 1200 mA / 51.3 mA would be ~15.6 hours. I could make it run longer if I had higher resistors.
