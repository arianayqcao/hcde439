        // include Servo library to control the servor motor
        #include 

        // create a servo object to control the servo motor
        Servo Servo1;

        // pin where voltage divider output (V-measure) is read
        const int sensorPin = A0;
        // pin connected to servo
        const int servoPin = 9;
        // PWM pin connected to LED
        const int ledPin = 6;

        // Lowest analogRead value I expect (dark room, ~0.45V)
        const int sensorMin = 150;
        // Highest analogRead value I expect (bright room, ~4.55V)
        const int sensorMax = 920;
        const int darkThreshold = 400;

        // setup code runs once at the start
        void setup() {
          // start serial communication at 9600 baud
          Serial.begin(9600);
          // set LED pin as output
          pinMode(ledPin, OUTPUT);
          // attach servo object to servo pin
          Servo1.attach(servoPin);
        }

        // main code runs repeatedly
        void loop() {
          // read voltage at divider
          int sensorValue = analogRead(sensorPin);
          // keep reading inside my calibrated range so map() doesn't overshoot
          sensorValue = constrain(sensorValue, sensorMin, sensorMax);
          // scale the reading to a servo angle: dark = 0 degrees, bright = 180 degrees
          int angle = map(sensorValue, sensorMin, sensorMax, 0, 180);

          int brightness = map(sensorValue, sensorMin, sensorMax, 255, 0);

          // move servo needle to the angle
          Servo1.write(angle);

          // set LED brightness using PWM
          analogWrite(ledPin, brightness);

          // check if room is dark enough to count as nighttime
          if (sensorValue < darkThreshold) {
            // report room is dark so lamp is glowing
            Serial.print("room is dark, lamp on | ");
          } else {
            // report room is bright so lamp is fading
            Serial.print("room is bright, lamp fading | ");
          }

          // label the raw sensor reading
          Serial.print("sensor: ");
          // print the raw sensor reading
          Serial.print(sensorValue);
          // label the servo angle
          Serial.print(" | needle: ");
          // print the servo angle
          Serial.print(angle);
          // label the LED brightness
          Serial.print(" deg | LED: ");
          // print the PWM value and end the line
          Serial.println(brightness);

          // wait 200 ms so the servo can move
          delay(200);

        }
